# Fuzz harnesses (libFuzzer entry points). Always built as gcc-compatible replay drivers that run in ctest under the
# sanitizers; with clang and -DG5C_FUZZ=ON they are also built as real coverage-guided fuzzers.
option(G5C_FUZZ "Build libFuzzer targets (needs clang)" OFF)
set(FUZZ_DIR ${CMAKE_SOURCE_DIR}/tests/fixtures)

function(g5c_fuzz name src libs)
  add_executable(replay_${name} ${src} replay_main.c)
  target_link_libraries(replay_${name} ${libs})
  if(G5C_FUZZ AND CMAKE_C_COMPILER_ID MATCHES "Clang")
    add_executable(fuzz_${name} ${src})
    target_compile_options(fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_libraries(fuzz_${name} ${libs})
  endif()
endfunction()
